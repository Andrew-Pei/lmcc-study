你来到小学帮助小朋友学数学。孩子们每天都会来问你各种数学题。小朋友们写题目时非常"自由"——有的写正常的+、-，有的画奇怪的符号（⊕、－、➕、➖），甚至用中文"加""减"。

def build_system_prompt() -> str:
    """
    定义 system prompt
    - 返回一个 system prompt，要求模型以"[Answer]: "的单行格式给出最终数值。
    """
    return """你是一个精确的数学计算助手。请按照以下步骤严格计算并返回准确的最终数值。

【计算步骤】
步骤1 - 符号转换：
   - 将问题中的所有符号转换为标准数学符号
   - ＋, ➕, ⊕, 加 → +
   - －, 减 → -
   - ×, *, 乘 → ×
   - ÷, /, 除 → ÷
   - ⊕ 不是异或运算，就是普通加法 +

步骤2 - 分步计算：
   - 按照运算优先级逐步计算

步骤3 - 输出答案：
   - 使用格式：[Answer]: 数值
   - 整数直接输出，确保计算准确
"""

def apply_chat_template_single(
    tokenizer: AutoTokenizer,
    system_prompt: str,
    problem: str,
) -> str:
    messages = [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": problem},
    ]
    rendered = tokenizer.apply_chat_template(
        messages,
        tokenize=False,
        add_generation_prompt=True,
        enable_thinking=True,  # 启用思考模式
    )
    return rendered

函数3：generate_single() - 单条推理

Pythonsubmission.py
def generate_single(
    model: AutoModelForCausalLM,
    tokenizer: AutoTokenizer,
    rendered_text: str,
    max_new_tokens: int,
    do_sample: bool,
) -> torch.Tensor:
    # 1. tokenize 输入
    inputs = tokenizer(rendered_text, return_tensors="pt", padding=True).to(model.device)

    # 2. 生成输出
    outputs = model.generate(
        input_ids=inputs["input_ids"],
        attention_mask=inputs.get("attention_mask"),
        max_new_tokens=max_new_tokens,
        do_sample=do_sample,
        eos_token_id=tokenizer.eos_token_id,
        pad_token_id=tokenizer.pad_token_id,
    )

    return outputs

函数4：generate_batch() - 批量推理

Pythonsubmission.py
def generate_batch(
    model: AutoModelForCausalLM,
    tokenizer: AutoTokenizer,
    rendered_texts: List[str],
    max_new_tokens: int,
    do_sample: bool,
) -> List[torch.Tensor]:
    all_outputs = []
    batch_size: int = 8

    for i in range(0, len(rendered_texts), batch_size):
        batch_texts = rendered_texts[i:i + batch_size]

        # 1. 批量 tokenize（padding + truncation）
        batch_inputs = tokenizer(
            batch_texts,
            return_tensors="pt",
            padding=True,        # 补齐到最长序列
            truncation=True,     # 截断超长序列
        ).to(model.device)

        # 2. 批量生成
        batch_outputs = model.generate(
            input_ids=batch_inputs["input_ids"],
            attention_mask=batch_inputs.get("attention_mask"),
            max_new_tokens=max_new_tokens,
            do_sample=do_sample,
            eos_token_id=tokenizer.eos_token_id,
            pad_token_id=tokenizer.pad_token_id,
        )

        all_outputs.append(batch_outputs)

    return all_outputs
!
批量推理注意事项

1. 必须设置padding=True，否则不同长度输入无法组成批次
2. 必须设置attention_mask，让模型忽略padding位置
3. 建议设置truncation=True防止超长序列
4. 分批处理时，batch_size根据显存调整
5. 返回原始输出，不做后处理！

核心任务1：compute_similarity()

使用给定的Embedding模型计算两个文本的余弦相似度。技术要点：Last-token pooling、L2 normalization、余弦相似度。

Pythonsubmission.py
import torch
import torch.nn.functional as F
from transformers import AutoTokenizer, AutoModel

def _last_token_pool(last_hidden_states, attention_mask):
    """
    Last-token pooling - 提取最后一个token的hidden state作为句向量

    参数：
        last_hidden_states: (batch_size, seq_len, hidden_dim)
        attention_mask: (batch_size, seq_len)
    返回：
        句向量 (batch_size, hidden_dim)
    """
    # 检查是否使用了左侧 padding
    left_padding = (attention_mask[:, -1].sum() == attention_mask.shape[0])

    if left_padding:
        # 左侧 padding：最后一个token都是有效的，直接取
        return last_hidden_states[:, -1]
    else:
        # 右侧 padding：需要找到每个样本最后一个有效token
        seq_lens = attention_mask.sum(dim=1) - 1
        batch_size = last_hidden_states.shape[0]
        return last_hidden_states[torch.arange(batch_size, device=last_hidden_states.device), seq_lens]


def compute_similarity(text1, text2, model, tokenizer):
    """
    计算两个文本之间的余弦相似度
    """
    device = next(model.parameters()).device

    # 1. 对文本进行分词和编码
    inputs = tokenizer(
        [text1, text2],
        padding=True,
        truncation=True,
        max_length=8192,
        return_tensors="pt",
    )
    inputs = {k: v.to(device) for k, v in inputs.items()}

    # 2. 获取模型输出
    with torch.no_grad():
        outputs = model(**inputs)

    # 3. Last-token pooling 提取句向量
    last_hidden_states = outputs.last_hidden_state
    sentence_embeddings = _last_token_pool(last_hidden_states, inputs["attention_mask"])

    # 4. L2 归一化（向量为单位向量）
    sentence_embeddings = F.normalize(sentence_embeddings, p=2, dim=1)

    # 5. 余弦相似度（归一化后点积=余弦相似度）
    similarity = (sentence_embeddings[0] @ sentence_embeddings[1]).item()

    return similarity

关键技术点详解

Last-token pooling vs Mean pooling：Qwen3-Embedding使用last-token pooling，因为因果语言模型的最后一个token包含了整个序列的信息。mean pooling会稀释信息。

左侧padding vs 右侧padding：左侧padding时，有效token在右侧，最后一个位置是有效的；右侧padding时，padding在右侧，需要找到最后一个有效token的位置。

L2归一化：F.normalize(x, p=2, dim=1)将向量除以其L2范数，长度变为1。归一化后点积=余弦相似度，值域[-1,1]。

相似度计算常见错误

1. 使用mean pooling而非last-token pooling → 相似度不匹配
2. 忘记L2归一化 → 点积不等于余弦相似度
3. 没有处理padding方向 → 取错token
4. 忘记torch.no_grad() → 浪费显存
5. 没有将输入移到模型所在设备 → 运行时错误

数据生成代码思路

可以使用Python脚本生成数据。核心思路：
1. 枚举运算符列表和数字范围
2. 随机组合数字和运算符生成题目
3. 计算正确答案
4. 过滤答案不在100-999范围的
5. 确保题目唯一
6. 生成超过1024条后计算相似度，去除相似度过高的，保留1024条

Pythongenerate_data.py (辅助脚本)
import json
import random
import itertools

# 运算符池
ADD_OPS = ['+', '＋', '➕', '⊕', '加']
SUB_OPS = ['-', '－', '➖', '减']
MUL_OPS = ['×', '*', '乘']
DIV_OPS = ['÷', '/', '除']

ALL_OPS = ADD_OPS + SUB_OPS + MUL_OPS + DIV_OPS

def compute(problem):
    """计算数学题的答案"""
    # 将各种符号转换为标准运算符
    s = problem
    for op in ADD_OPS:
        s = s.replace(op, '+')
    for op in SUB_OPS:
        s = s.replace(op, '-')
    for op in MUL_OPS:
        s = s.replace(op, '*')
    for op in DIV_OPS:
        s = s.replace(op, '/')
    # 替换中文括号
    s = s.replace('【', '(').replace('】', ')')
    # 评估表达式
    try:
        result = eval(s)
        if isinstance(result, float) and result == int(result):
            result = int(result)
        return result
    except:
        return None

def generate_problems(n=1024):
    problems = set()
    seen_answers = set()

    while len(problems) < n * 3:  # 多生成一些用于筛选
        # 随机选择1-3个数字
        n_nums = random.choice([1, 2, 3])
        nums = [random.randint(1, 999) for _ in range(n_nums)]

        # 随机选择运算符
        ops = [random.choice(ALL_OPS) for _ in range(n_nums - 1)]

        # 构建题目
        problem = str(nums[0])
        for i in range(len(ops)):
            problem += f' {ops[i]} {nums[i+1]}'

        # 计算答案
        answer = compute(problem)

        # 过滤
        if answer is None:
            continue
        if isinstance(answer, float):
            continue
        if answer < 100 or answer > 999:
            continue
        if problem in problems:
            continue

        problems.add(problem)

    # 转换为JSONL格式
    results = []
    for p in list(problems)[:n]:
        a = compute(p)
        results.append({"problem": p, "answer": str(a)})

    return results

# 保存到文件
data = generate_problems(1024)
with open("data/dataset.jsonl", "w", encoding="utf-8") as f:
    for item in data:
        f.write(json.dumps(item, ensure_ascii=False) + "\n")
